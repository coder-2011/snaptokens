+use std::{
+    cell::UnsafeCell,
+    marker::PhantomData,
+    ops::{Deref, DerefMut},
+    sync::atomic::{AtomicU8, Ordering},
+};
+
+pub(super) struct SpinMutex<T> {
+    state: AtomicU8,
+    value: UnsafeCell<T>,
+}
+
+// Only the successful Acquire CAS grants access to value; guard drop publishes
+// its writes with Release before another thread can acquire that same value.
+unsafe impl<T: Send> Sync for SpinMutex<T> {}
+
+impl<T> SpinMutex<T> {
+    pub(super) fn new(value: T) -> Self {
+        Self {
+            state: AtomicU8::new(0),
+            value: UnsafeCell::new(value),
+        }
+    }
+
+    pub(super) fn lock(&self) -> Guard<'_, T> {
+        loop {
+            match self
+                .state
+                .compare_exchange_weak(0, 1, Ordering::Acquire, Ordering::Relaxed)
+            {
+                Ok(_) => {
+                    return Guard {
+                        lock: self,
+                        marker: PhantomData,
+                        panicking: std::thread::panicking(),
+                    };
+                }
+                Err(2) => panic!("shared cache lock poisoned"),
+                Err(_) => {
+                    while self.state.load(Ordering::Relaxed) == 1 {
+                        std::hint::spin_loop();
+                    }
+                }
+            }
+        }
+    }
+}
+
+pub(super) struct Guard<'a, T> {
+    lock: &'a SpinMutex<T>,
+    marker: PhantomData<&'a mut T>,
+    panicking: bool,
+}
+
+impl<T> Deref for Guard<'_, T> {
+    type Target = T;
+    fn deref(&self) -> &T {
+        // The guard holds exclusive access until its Release store.
+        unsafe { &*self.lock.value.get() }
+    }
+}
+
+impl<T> DerefMut for Guard<'_, T> {
+    fn deref_mut(&mut self) -> &mut T {
+        // A mutable guard borrow excludes every other access through this guard.
+        unsafe { &mut *self.lock.value.get() }
+    }
+}
+
+impl<T> Drop for Guard<'_, T> {
+    fn drop(&mut self) {
+        let poisoned = !self.panicking && std::thread::panicking();
+        self.lock
+            .state
+            .store(if poisoned { 2 } else { 0 }, Ordering::Release);
+    }
+}

+    #[test]
+    fn mutual_exclusion_and_publication() {
+        let lock = Arc::new(SpinMutex::new((0u64, 0u64)));
+        std::thread::scope(|scope| {
+            for _ in 0..8 {
+                let lock = &lock;
+                scope.spawn(move || {
+                    for _ in 0..25_000 {
+                        let mut value = lock.lock();
+                        assert_eq!(value.0, value.1);
+                        value.0 += 1;
+                        if value.0 % 257 == 0 {
+                            std::thread::yield_now();
+                        }
+                        value.1 += 1;
+                    }
+                });
+            }
+        });
+        assert_eq!(*lock.lock(), (200_000, 200_000));
+    }
